cmake_minimum_required(VERSION 3.20)
project(lab1_sniffer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Widgets)
set(NPCAP_SDK_DIR "${CMAKE_SOURCE_DIR}/.deps/npcap-sdk-1.17" CACHE PATH "Npcap SDK root")
if(NOT EXISTS "${NPCAP_SDK_DIR}/Include/pcap.h")
  message(FATAL_ERROR "Npcap SDK missing: set NPCAP_SDK_DIR")
endif()
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The A01 validated baseline requires MSVC x64")
endif()
add_executable(lab1_env_probe probes/environment/probe.cpp)
target_include_directories(lab1_env_probe PRIVATE "${NPCAP_SDK_DIR}/Include")
target_link_libraries(lab1_env_probe PRIVATE Qt5::Widgets "${NPCAP_SDK_DIR}/Lib/x64/wpcap.lib" ws2_32 delayimp)
target_link_options(lab1_env_probe PRIVATE /DELAYLOAD:wpcap.dll)
