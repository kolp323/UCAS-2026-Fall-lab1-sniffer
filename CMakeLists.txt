cmake_minimum_required(VERSION 3.20)
project(lab1_sniffer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Widgets Network)
set(NPCAP_SDK_DIR "${CMAKE_SOURCE_DIR}/.deps/npcap-sdk-1.17" CACHE PATH "Npcap SDK root")
if(NOT EXISTS "${NPCAP_SDK_DIR}/Include/pcap.h")
  message(FATAL_ERROR "Npcap SDK missing: set NPCAP_SDK_DIR")
endif()
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Validated baseline requires MSVC x64")
endif()
add_compile_options(/utf-8 /W4)
add_library(lab1_parser src/parser.cpp src/display_filter.cpp src/tcp_flow.cpp)
target_link_libraries(lab1_parser PUBLIC Qt5::Core Qt5::Network)
target_include_directories(lab1_parser PUBLIC src)
function(capture_target target)
  target_include_directories(${target} PRIVATE "${NPCAP_SDK_DIR}/Include")
  target_link_libraries(${target} PRIVATE "${NPCAP_SDK_DIR}/Lib/x64/wpcap.lib" ws2_32 delayimp)
  target_link_options(${target} PRIVATE /DELAYLOAD:wpcap.dll)
endfunction()
add_executable(lab1_cli src/cli.cpp)
target_link_libraries(lab1_cli PRIVATE lab1_parser)
capture_target(lab1_cli)
add_library(lab1_process src/process_view.cpp)
target_link_libraries(lab1_process PUBLIC Qt5::Widgets Qt5::Network PRIVATE iphlpapi ws2_32)
add_executable(lab1_sniffer src/gui.cpp)
target_link_libraries(lab1_sniffer PRIVATE lab1_process lab1_parser Qt5::Widgets Qt5::Network)
capture_target(lab1_sniffer)
